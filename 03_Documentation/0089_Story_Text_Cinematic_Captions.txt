--------------------------------------------------

DRAW TEXT BOX
--------------------------------------------------
void DrawIntroCaption(const char *text)
{
    int boxHeight = 120;

    DrawRectangle(
        0,
        GetScreenHeight() - boxHeight,
        GetScreenWidth(),
        boxHeight,
        Fade(BLACK, 0.72f)
    );

    DrawText(
        text,
        40,
        GetScreenHeight() - 80,
        28,
        WHITE
    );
}
